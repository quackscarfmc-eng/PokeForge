   149→   149→function slug(name: string): string {
   150→   150→  return name
   151→   151→    .toLowerCase()
   152→   152→    .replace(/[^a-z0-9]+/g, "")
   153→   153→    .replace(/^(.)(.*)$/, (_, a, b) => a.toUpperCase() + b);
   154→   154→}
   155→   155→
   156→   156→function toPascalCase(s: string): string {
   157→   157→  return s
   158→   158→    .split(/[^a-zA-Z0-9]+/)
   159→   159→    .filter(Boolean)
   160→   160→    .map((w) => w.charAt(0).toUpperCase() + w.slice(1).toLowerCase())
   161→   161→    .join("");
   162→   162→}

   164→   164→function constantToFolder(constant: string): string {
   165→   165→  // SPECIES_MY_COOL_MON -> my_cool_mon

   167→   167→}

   169→   169→function compoundString(text: string, perLine = 22): string {
   170→   170→  const words = text.split(/\s+/);
   171→   171→  const lines: string[] = [];
   172→   172→  let cur = "";
   173→   173→  for (const w of words) {
   174→   174→    if ((cur + " " + w).trim().length > perLine) {
   175→   175→      lines.push(cur.trim());
   176→   176→      cur = w;
   177→   177→    } else {
   178→   178→      cur = (cur + " " + w).trim();
   179→   179→    }
   180→   180→  }
   181→   181→  if (cur) lines.push(cur.trim());
   182→   182→  return lines.map((l) => `_${(`"${l}\\n"`).padEnd(2)}`).join("\n    ");
   183→   183→}

   407→   407→    .join("\n");
   408→   408→
   409→   409→  return `// ============================================================
   410→   410→// ${t.constantName} (${t.name})
   411→   411→// Generated by PokeForge
   412→   412→// ============================================================
   413→   413→
   414→   414→// ---- 1. include/constants/types.h ----
   415→   415→#define ${t.constantName} ${t.typeId}
   416→   416→#define NUMBER_OF_TYPES (${t.typeId} + 1)
   417→   417→
   418→   418→// ---- 2. src/data/type_effectiveness.h (extend chart) ----
   419→   419→// Offensive row (what ${t.name} deals to other types):
   420→   420→${offensiveRows}
   421→   421→
   422→   422→// Defensive column (what ${t.name} takes from other types):
   423→   423→${defensiveRows}
   424→   424→
   425→   425→// ---- 3. (Optional) Update display name in src/data/types.c ----

   427→   427→
   428→   428→// ---- 4. (Optional) Add type icon + battle animation ----
   429→   429→// See data/battle_anim_scripts.s and graphics/types/
   430→   430→`;

   545→   545→export function validateConstantName(name: string, prefix: string): ValidationIssue[] {
   546→   546→  const issues: ValidationIssue[] = [];
   547→   547→  if (!name.startsWith(prefix))
   548→   548→    issues.push({ severity: "error", message: `Constant must start with "${prefix}"` });
   549→   549→  if (!/^[A-Z][A-Z0-9_]*$/.test(name))
   550→   550→    issues.push({ severity: "error", message: "Constant must be UPPER_SNAKE_CASE (A-Z, 0-9, _)" });
   551→   551→  if (name.length > 41)
   552→   552→    issues.push({ severity: "error", message: "Constant name must be ≤ 41 characters" });
   553→   553→  if (name.length < prefix.length + 2)
   554→   554→    issues.push({ severity: "warning", message: "Constant name is very short" });
   555→   555→  return issues;
   556→   556→}

   558→   558→export function validateSpecies(s: SpeciesData, knownTypes: string[], knownAbilities: string[], knownMoves: string[]): ValidationIssue[] {
   559→   559→  const issues: ValidationIssue[] = [];
   560→   560→  issues.push(...validateConstantName(s.constantName, "SPECIES_"));
   561→   561→
   562→   562→  if (s.speciesId < 0 || s.speciesId > 65535)
   563→   563→    issues.push({ severity: "error", message: "Species ID must be 0–65535" });
   564→   564→
   565→   565→  const statBounds: [keyof SpeciesData, number, number][] = [
   566→   566→    ["baseHP", 1, 255], ["baseAttack", 1, 255], ["baseDefense", 1, 255],
   567→   567→    ["baseSpeed", 1, 255], ["baseSpAttack", 1, 255], ["baseSpDefense", 1, 255],
   568→   568→  ];
   569→   569→  for (const [k, lo, hi] of statBounds) {
   570→   570→    const v = s[k] as number;
   571→   571→    if (v < lo || v > hi)
   572→   572→      issues.push({ severity: "error", message: `${k} must be ${lo}–${hi}` });
   573→   573→  }
   574→   574→  if (s.catchRate < 0 || s.catchRate > 255) issues.push({ severity: "error", message: "Catch rate must be 0–255" });
   575→   575→  if (s.expYield < 0 || s.expYield > 255) issues.push({ severity: "error", message: "EXP yield must be 0–255" });
   576→   576→  if (s.genderRatio < 0 || s.genderRatio > 255) issues.push({ severity: "error", message: "Gender ratio must be 0–255" });
   577→   577→  if (s.eggCycles < 0 || s.eggCycles > 255) issues.push({ severity: "error", message: "Egg cycles must be 0–255" });
   578→   578→  if (s.friendship < 0 || s.friendship > 255) issues.push({ severity: "error", message: "Friendship must be 0–255" });
   579→   579→
   580→   580→  if (s.types.length === 0) issues.push({ severity: "error", message: "Species must have at least 1 type" });
   581→   581→  if (s.types.length > 2) issues.push({ severity: "error", message: "Species can have at most 2 types" });
   582→   582→  for (const t of s.types) {
   583→   583→    if (!knownTypes.includes(t) && !POKEMON_TYPES.some((pt) => pt.constant === t))
   584→   584→      issues.push({ severity: "error", message: `Unknown type: ${t}` });
   585→   585→  }
   586→   586→
   587→   587→  if (s.abilities.length === 0 || s.abilities.length > 3)
   588→   588→    issues.push({ severity: "error", message: "Species must have 1–3 abilities" });
   589→   589→  for (const a of s.abilities) {
   590→   590→    if (!knownAbilities.includes(a) && a !== "ABILITY_NONE")
   591→   591→      issues.push({ severity: "error", message: `Unknown ability: ${a}` });
   592→   592→  }
   593→   593→
   594→   594→  if (!GROWTH_RATES.includes(s.growthRate)) issues.push({ severity: "error", message: `Unknown growth rate: ${s.growthRate}` });
   595→   595→  if (s.eggGroups.length === 0 || s.eggGroups.length > 2)
   596→   596→    issues.push({ severity: "error", message: "Must have 1–2 egg groups" });
   597→   597→  for (const eg of s.eggGroups) {
   598→   598→    if (!EGG_GROUPS.includes(eg))
   599→   599→      issues.push({ severity: "error", message: `Unknown egg group: ${eg}` });
   600→   600→  }
   601→   601→  if (!BODY_COLORS.includes(s.bodyColor)) issues.push({ severity: "error", message: `Unknown body color: ${s.bodyColor}` });
   602→   602→  if (!CRY_IDS.includes(s.cryId)) issues.push({ severity: "warning", message: `Cry ${s.cryId} not in known list (may still be valid)` });
   603→   603→  if (!FRONT_ANIM_IDS.includes(s.frontAnimId)) issues.push({ severity: "warning", message: `Unknown frontAnimId: ${s.frontAnimId}` });
   604→   604→  if (!BACK_ANIM_IDS.includes(s.backAnimId)) issues.push({ severity: "warning", message: `Unknown backAnimId: ${s.backAnimId}` });
   605→   605→
   606→   606→  if (s.speciesName.length === 0) issues.push({ severity: "error", message: "Species name is required" });
   607→   607→  if (s.speciesName.length > 11) issues.push({ severity: "warning", message: "Species name > 11 chars may overflow battle UI" });
   608→   608→  if (s.height <= 0) issues.push({ severity: "warning", message: "Height should be > 0 (decimetres)" });
   609→   609→  if (s.weight <= 0) issues.push({ severity: "warning", message: "Weight should be > 0 (hectograms)" });
   610→   610→
   611→   611→  if (!s.description || s.description.length < 10)
   612→   612→    issues.push({ severity: "warning", message: "Description is short or missing" });
   613→   613→
   614→   614→  return issues;
   615→   615→}

   617→   617→export function validateMove(m: MoveData, knownTypes: string[]): ValidationIssue[] {
   618→   618→  const issues: ValidationIssue[] = [];
   619→   619→  issues.push(...validateConstantName(m.constantName, "MOVE_"));
   620→   620→  if (m.moveId < 0 || m.moveId > 65535) issues.push({ severity: "error", message: "Move ID must be 0–65535" });
   621→   621→  if (m.power < 0 || m.power > 255) issues.push({ severity: "error", message: "Power must be 0–255" });
   622→   622→  if (m.accuracy < 0 || m.accuracy > 100) issues.push({ severity: "error", message: "Accuracy must be 0–100" });
   623→   623→  if (m.pp < 1 || m.pp > 40) issues.push({ severity: "warning", message: "PP is typically 1–40" });
   624→   624→  if (m.priority < -7 || m.priority > 5) issues.push({ severity: "error", message: "Priority must be -7…+5" });
   625→   625→  if (m.critStage < 0 || m.critStage > 24) issues.push({ severity: "error", message: "Crit stage must be 0–24" });
   626→   626→
   627→   627→  if (!POKEMON_TYPES.some((t) => t.constant === m.type) && !knownTypes.includes(m.type))
   628→   628→    issues.push({ severity: "error", message: `Unknown type: ${m.type}` });
   629→   629→  if (!MOVE_EFFECTS.some((e) => e.constant === m.effect)) issues.push({ severity: "warning", message: `Effect ${m.effect} may be unknown` });
   630→   630→  if (!MOVE_CATEGORIES.some((c) => c.constant === m.category)) issues.push({ severity: "error", message: `Unknown category: ${m.category}` });
   631→   631→  if (!MOVE_TARGETS.some((t) => t.constant === m.target)) issues.push({ severity: "error", message: `Unknown target: ${m.target}` });
   632→   632→
   633→   633→  if (!m.name) issues.push({ severity: "error", message: "Move name is required" });
   634→   634→  if (m.name.length > 12) issues.push({ severity: "warning", message: "Move name > 12 chars may overflow battle UI" });
   635→   635→  if (m.category === "CATEGORY_STATUS" && m.power !== 0) issues.push({ severity: "warning", message: "Status moves usually have power 0" });
   636→   636→  if (m.category !== "CATEGORY_STATUS" && m.power === 0) issues.push({ severity: "warning", message: "Damaging move with 0 power" });
   637→   637→
   638→   638→  return issues;
   639→   639→}

   641→   641→export function validateType(t: TypeData, knownTypes: string[]): ValidationIssue[] {
   642→   642→  const issues: ValidationIssue[] = [];
   643→   643→  issues.push(...validateConstantName(t.constantName, "TYPE_"));
   644→   644→  if (t.typeId < 0 || t.typeId > 255) issues.push({ severity: "error", message: "Type ID must be 0–255" });
   645→   645→  if (!t.name) issues.push({ severity: "error", message: "Type name is required" });
   646→   646→
   647→   647→  const allTypes = [...POKEMON_TYPES.map((pt) => pt.constant), ...knownTypes];
   648→   648→  for (const [atk, mult] of Object.entries(t.offensiveMatrix)) {
   649→   649→    if (!allTypes.includes(atk) && atk !== t.constantName)
   650→   650→      issues.push({ severity: "warning", message: `Offensive matrix references unknown type: ${atk}` });
   651→   651→    if (![0, 0.25, 0.5, 1, 2, 4].includes(mult))
   652→   652→      issues.push({ severity: "warning", message: `Offensive mult for ${atk} is unusual: ${mult}` });
   653→   653→  }
   654→   654→  for (const [atk, mult] of Object.entries(t.defensiveMatrix)) {
   655→   655→    if (!allTypes.includes(atk) && atk !== t.constantName)
   656→   656→      issues.push({ severity: "warning", message: `Defensive matrix references unknown type: ${atk}` });
   657→   657→    if (![0, 0.25, 0.5, 1, 2, 4].includes(mult))
   658→   658→      issues.push({ severity: "warning", message: `Defensive mult for ${atk} is unusual: ${mult}` });
   659→   659→  }
   660→   660→
   661→   661→  if (!/^#[0-9A-Fa-f]{6}$/.test(t.colorHex)) issues.push({ severity: "warning", message: "colorHex should be #RRGGBB" });
   662→   662→  return issues;
   663→   663→}

   716→   716→export function planToMarkdown(p: ChangePlan): string {
   717→   717→  const lines: string[] = [];
   718→   718→  lines.push("# Change Plan");
   719→   719→  lines.push("");
   720→   720→  lines.push(`- **Mode:** \`${p.mode}\``);
   721→   721→  lines.push(`- **Entity type:** \`${p.entityType}\``);
   722→   722→  lines.push(`- **Entity constant:** \`${p.entityConstant}\``);
   723→   723→  lines.push(`- **Blocked:** \`${p.isBlocked ? "YES" : "no"}\``);
   724→   724→  lines.push(`- **Steps:** ${p.steps.length}`);
   725→   725→  lines.push("");
   726→   726→  if (p.errors.length) {
   727→   727→    lines.push("## ❌ Errors (apply blocked)");
   728→   728→    p.errors.forEach((e) => lines.push(`- ${e}`));
   729→   729→    lines.push("");
   730→   730→  }
   731→   731→  if (p.warnings.length) {
   732→   732→    lines.push("## ⚠️ Warnings");
   733→   733→    p.warnings.forEach((w) => lines.push(`- ${w}`));
   734→   734→    lines.push("");
   735→   735→  }
   736→   736→  if (p.steps.length) {
   737→   737→    lines.push("## 📝 Steps");
   738→   738→    p.steps.forEach((s, i) => {
   739→   739→      lines.push(`### ${i + 1}. \`${s.action}\` — \`${s.targetFile}\``);
   740→   740→      lines.push(`- **Reason:** ${s.reason}`);
   741→   741→      lines.push(`- **Risk:** \`${s.riskLevel}\``);
   742→   742→      if (s.warnings.length) s.warnings.forEach((w) => lines.push(`  - ⚠️ ${w}`));
   743→   743→      lines.push("");
   744→   744→    });
   745→   745→  }
   746→   746→  if (p.generatedCode) {
   747→   747→    lines.push("## 📄 Generated code (preview)");
   748→   748→    lines.push("```c");
   749→   749→    lines.push(p.generatedCode);
   750→   750→    lines.push("```");
   751→   751→  }
   752→   752→  return lines.join("\n");
   753→   753→}
